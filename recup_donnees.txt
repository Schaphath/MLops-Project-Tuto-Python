SELECT * FROM predictions;

SELECT
    created_at,
    prediction,
    probability_pct,
    "perimeter",
    "concave_points",
    "texture",
    "smoothness",
    "symmetry"
FROM predictions
WHERE user_id = 1
ORDER BY created_at DESC;
