# Créer une branche
git branch modification_api
git checkout modification_api

# Revenir à la branche d'origine 
git checkout main 
git pull origin main
git merge modification_api
git push origin main 

# supprimer la branche 
git branch -d modification_api

# Gestion de conflit
git pull origin main --rebase
git push origin main
git push origin main --force


# 1. Sauvegarder les modifications dans le stash
git stash

# 2. Basculer sur main
git switch main

# 3. (Quand tu reviendras sur codemodif) Récupérer tes modifications :
# git switch codemodif
# git stash pop
